  ensures  result >= 0.0_joule
= ...

pub struct RingWindow<T, comptime CAP: usize = 64> {
  pub buf: [CAP]T
  pub head: usize[0..CAP] = 0
  pub len: usize[0..=CAP] = 0

  invariant {
    CAP > 0
    head < CAP
    len <= CAP
  }
}
